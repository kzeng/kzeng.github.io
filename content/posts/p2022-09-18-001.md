# 在oangepi zero2 上实验通过
import os
import time
import configparser

# 采用wpi编号， 对应orangepi zero2 的pin 7 (gpio)
FAN_GPIO = 2
os.system('gpio mode '+str(FAN_GPIO)+' out')

while True:
	# temp_hight = 50
	# temp_low =49 
	# monitor_interval = 10
	# opi_fan_auto = off
    # 以上参数亦可来自某个配置文件，如下所示
	cf = configparser.ConfigParser()
	cf.read("Config.ini")
	temp_hight = int(cf.get("fan", "temp_hight"))
	temp_low = int(cf.get("fan", "temp_low"))
	monitor_interval = int(cf.get("fan", "monitor_interval"))
	opi_fan_auto = cf.get("fan", "opi_fan_auto")

	if opi_fan_auto == "on":
	    # get cpu temp
		temp=os.popen('cat /sys/class/thermal/thermal_zone0/temp').readlines()[0]
		cpu_temp=int(temp)/1000

		#print("cpu temp: "+str(cpu_temp))
		current_time = time.strftime("%Y-%m-%d %H:%M:%S",time.localtime())

		if cpu_temp >= temp_hight:
			os.system('gpio write '+str(FAN_GPIO)+' 0')
			print("Time:{}\tCPU Temp:{}\t\tFan Status:{}".format(current_time, cpu_temp, 'Running'))

		if cpu_temp <= temp_low:
			os.system('gpio write '+str(FAN_GPIO)+' 1')
			print("Time:{}\tCPU Temp:{}\t\tFan Status:{}".format(current_time, cpu_temp, 'Stopped'))

		time.sleep(monitor_interval)
