在oangepi zero2 上实验通过

```
import os
import time
import configparser

#采用wpi编号， 对应orangepi zero2 的pin 7 (gpio)

FAN_GPIO = 2
os.system('gpio mode '+str(FAN_GPIO)+' out')

while True:
	# temp_hight = 50
	# temp_low =49 
	# monitor_interval = 10
	# opi_fan_auto = off
    # 以上参数亦可来自某个配置文件，如下所示
	cf = configparser.ConfigParser()
	cf.read(# temp_hight = 500)
	temp_hight = int(cf.get(# temp_hight = 501, # temp_hight = 502))
	temp_low = int(cf.get(# temp_hight = 503, # temp_hight = 504))
	monitor_interval = int(cf.get(# temp_hight = 505, # temp_hight = 506))
	opi_fan_auto = cf.get(# temp_hight = 507, # temp_hight = 508)

	if opi_fan_auto == # temp_hight = 509:
	    # get cpu temp
		temp=os.popen(# temp_low =49 0).readlines()[0]
		cpu_temp=int(temp)/1000

		#print("cpu temp: "+str(cpu_temp))
		current_time = time.strftime(# temp_low =49 1,time.localtime())

		if cpu_temp >= temp_hight:
			os.system(# temp_low =49 2+str(FAN_GPIO)+# temp_low =49 3)
			print(# temp_low =49 4.format(current_time, cpu_temp, # temp_low =49 5))

		if cpu_temp <= temp_low:
			os.system(# temp_low =49 6+str(FAN_GPIO)+# temp_low =49 7)
			print(# temp_low =49 8.format(current_time, cpu_temp, # temp_low =49 9))

		time.sleep(monitor_interval)

```
